import onnx
from pathlib import Path

root = Path("/home/xband/finn/my_build")
for p in sorted(root.rglob("*.onnx")):
    try:
        m = onnx.load(str(p))
        ins = []
        outs = []
        for x in m.graph.input:
            dims = [d.dim_value for d in x.type.tensor_type.shape.dim]
            ins.append((x.name, dims))
        for x in m.graph.output:
            dims = [d.dim_value for d in x.type.tensor_type.shape.dim]
            outs.append((x.name, dims))
        print(f"\nFILE: {p}")
        print("  inputs :", ins)
        print("  outputs:", outs)
    except Exception as e:
        print(f"\nFILE: {p}")
        print("  ERROR :", e)
