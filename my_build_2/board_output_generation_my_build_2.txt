This document assumes:

- build directory: `/home/xband/finn/my_build_2`
- deploy directory: `/home/xband/finn/my_build_2/output_u250_bitfile_2/deploy/driver`
- compact-complex output layout
- input shape: `(1, 256)`

Paste this code:

```cpp
#include <fstream>
#include <iostream>
#include <vector>
#include <cstring>
#include <stdexcept>
#include <cstdint>

#include <xrt/xrt_bo.h>
#include <xrt/xrt_device.h>
#include <xrt/xrt_kernel.h>

static std::vector<char> read_binary_file(const std::string& filename) {
    std::ifstream ifs(filename, std::ios::binary | std::ios::ate);
    if (!ifs) {
        throw std::runtime_error("Cannot open file: " + filename);
    }

    std::streamsize size = ifs.tellg();
    if (size < 0) {
        throw std::runtime_error("Failed to get file size: " + filename);
    }

    ifs.seekg(0, std::ios::beg);
    std::vector<char> buffer(static_cast<size_t>(size));
    if (!ifs.read(buffer.data(), size)) {
        throw std::runtime_error("Failed to read file: " + filename);
    }
    return buffer;
}

int main() {
    try {
        const std::string xclbin_path = "../bitfile/finn-accel.xclbin";
        const std::string input_path = "input.bin";
        const std::string output_path = "output.bin";

        const size_t input_bytes = 1 * 256 * 1;
        const size_t output_elems = 1 * 256;
        const size_t bytes_per_output_elem = 2;
        const size_t output_bytes = output_elems * bytes_per_output_elem;
        const uint32_t numReps = 1;

        std::cout << "Opening device 0..." << std::endl;
        auto device = xrt::device(0);

        std::cout << "Loading xclbin: " << xclbin_path << std::endl;
        auto uuid = device.load_xclbin(xclbin_path);

        std::cout << "Opening kernels..." << std::endl;
        auto idma = xrt::kernel(device, uuid, "StreamingDataflowPartition_0");
        auto odma = xrt::kernel(device, uuid, "StreamingDataflowPartition_2");

        std::cout << "Allocating buffers..." << std::endl;
        auto in_bo = xrt::bo(device, input_bytes, idma.group_id(0));
        auto out_bo = xrt::bo(device, output_bytes, odma.group_id(0));

        auto in_map = in_bo.map<uint8_t*>();
        auto out_map = out_bo.map<uint8_t*>();

        std::cout << "Reading input.bin..." << std::endl;
        auto input_data = read_binary_file(input_path);
        if (input_data.size() != input_bytes) {
            std::cerr << "Input size mismatch. Expected "
                      << input_bytes
                      << " bytes, got "
                      << input_data.size()
                      << " bytes."
                      << std::endl;
            return 1;
        }

        std::memcpy(in_map, input_data.data(), input_bytes);
        std::memset(out_map, 0, output_bytes);

        std::cout << "Sync input buffer to device..." << std::endl;
        in_bo.sync(XCL_BO_SYNC_BO_TO_DEVICE);
        out_bo.sync(XCL_BO_SYNC_BO_TO_DEVICE);

        std::cout << "Starting output DMA first..." << std::endl;
        auto odma_run = odma(out_bo, numReps);

        std::cout << "Starting input DMA..." << std::endl;
        auto idma_run = idma(in_bo, numReps);

        std::cout << "Waiting for IDMA..." << std::endl;
        idma_run.wait();

        std::cout << "Waiting for ODMA..." << std::endl;
        odma_run.wait();

        std::cout << "Sync output buffer from device..." << std::endl;
        out_bo.sync(XCL_BO_SYNC_BO_FROM_DEVICE);

        std::cout << "Writing output.bin..." << std::endl;
        std::ofstream ofs(output_path, std::ios::binary);
        if (!ofs) {
            throw std::runtime_error("Cannot open output file: " + output_path);
        }

        ofs.write(
            reinterpret_cast<char*>(out_map),
            static_cast<std::streamsize>(output_bytes)
        );
        ofs.close();

        std::cout << "Done. Output saved to " << output_path << std::endl;
        std::cout << "Expected input bytes: " << input_bytes << std::endl;
        std::cout << "Expected output tensor shape: (1, 256)" << std::endl;
        std::cout << "Expected output layout: first 128 real, last 128 imag" << std::endl;
        std::cout << "Expected output datatype: INT16" << std::endl;
        std::cout << "Expected output bytes: " << output_bytes << std::endl;
        return 0;
    } catch (const std::exception& e) {
        std::cerr << "ERROR: " << e.what() << std::endl;
        return 1;
    }
}
```

Step 5: quick check of `output.bin` and decode it as `(1, 256)`

```bash
cd /home/xband/finn/my_build_2/output_u250_bitfile_2/deploy/driver
python3 - <<'PY'
from pathlib import Path
import numpy as np

p = Path("output.bin")
if not p.is_file():
    raise FileNotFoundError("output.bin not found")

raw = np.fromfile("output.bin", dtype=np.uint8)
print("raw bytes:", raw.size)
if raw.size != 512:
    raise RuntimeError(f"Expected 512 bytes for INT16 (1,256), got {raw.size}")

y_raw = np.fromfile("output.bin", dtype="<i2").astype(np.int32).reshape(1, 256)
np.save("output_raw_int16_1x256.npy", y_raw)
np.save("output.npy", y_raw.astype(np.float32))

print("Saved output_raw_int16_1x256.npy")
print("Saved output.npy")
print("output_raw shape:", y_raw.shape)
print("output_raw dtype:", y_raw.dtype)
print("output_raw min/max:", y_raw.min(), y_raw.max())
print("real first 16:", y_raw[0, :16])
print("imag first 16:", y_raw[0, 128:144])
print("real nonzero:", np.count_nonzero(y_raw[0, :128]))
print("imag nonzero:", np.count_nonzero(y_raw[0, 128:]))
PY
```

Step 6: dequantize on the host machine

After copying `output.bin` or `output_raw_int16_1x256.npy` back into this
repository, run:

```powershell
python dequantize_board_output.py --model my_build_2/deeponet_u250_int4_qonnx.onnx --input my_build_2/verification/case_000/output_raw_int16_1x256.npy
```

For all verification cases at once:

```powershell
python dequantize_board_output.py --model my_build_2/deeponet_u250_int4_qonnx.onnx --cases_dir my_build_2/verification --input_name output_raw_int16_1x256.npy
```

This writes `output_dequant.npy`, which can then be passed directly to
`step6_compare_deploy_output.py`.

Step 7: send the result back to the host machine

Prefer sending:

- `output.bin`

If convenient, also send:

- `output_raw_int16_1x256.npy`
- `output.npy`

The host side can then dequantize and validate the result again using the
`(1,256)` compact-complex layout.
